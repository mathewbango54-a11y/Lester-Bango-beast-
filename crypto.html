${esc(t.pair)} ${money(t.entry)}${money(t.exit)} ${money(t.pnl)} ${esc(t.reason)} `).join("") || 'No closed trades yet'; } function paperBuy(key) { rollDay(); const m = latestMarkets.find(x => x.key === key); const s = settings(); if (!m) return status("Market not found. No real crypto was purchased."); } function paperSell(id,reason) { const i = account.open.findIndex(t => t.id === id); if (i < 0) return; const t = account.open[i]; const exit = t.last; const proceeds = t.qty*exit*(1-t.fee); const pnl = proceeds-t.stake; account.cash += proceeds; account.history.push({ exchange:t.exchange, pair:t.pair, entry:t.entry, exit, pnl, reason, day:account.day, opened:t.opened, closed:new Date().toISOString() }); account.open.splice(i,1); save(); render(); status("Paper position closed: "+reason+ ".
